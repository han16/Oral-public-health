compute_survival_fast <- function(claims_dt, demo_dt, start_code, end_code ) {
  
  # 1. Find the first occurrence of the start_code for each tooth
  start_claims <- claims_dt[`Submitted Procedure Code` == start_code, 
                            .(Start_Date = min(`Service Date`)), 
                            by = .(Tooth_ID = member_tooth_number, `Unique Member ID`)]
  
  if (nrow(start_claims) == 0) return(data.table())
  
  # 2. Find the first occurrence of the end event (7140) for each tooth
  end_claims <- claims_dt[`Submitted Procedure Code` == end_code, 
                          .(End_Event_Date = min(`Service Date`)), 
                          by = .(Tooth_ID = member_tooth_number)]
  
  # 3. Pull metadata (specialty, surface area) for the tooth's initial start claim
  # (Using data.table's fast .SD syntax to grab the first row attributes)
  meta_claims <- claims_dt[`Submitted Procedure Code` == start_code, 
                           .(provider_specialty = `Provider Specialty Name`[1],
                             surface_area_code = `All Surfaces Area Code`[1]), 
                           by = .(Tooth_ID = member_tooth_number)]
  
  # 4. Merge start claims with demo to get member termination date
  setkey(demo_dt, `Unique Member ID`)
  setkey(start_claims, `Unique Member ID`)
  merged <- demo_dt[start_claims, nomatch = 0]
  
  # 5. Merge with end events and metadata
  setkey(merged, Tooth_ID)
  setkey(end_claims, Tooth_ID)
  merged <- end_claims[merged] # left join
  
  setkey(meta_claims, Tooth_ID)
  merged <- meta_claims[merged]
  
  # 6. Compute survival metrics vectorized (no loops!)
  merged[, Termination_Date := as.Date(`Member Termination Date`)]
  merged[, Event_Observed := ifelse(!is.na(End_Event_Date) & End_Event_Date >= Start_Date, 1, 0)]
  
  # End date is min of (Extraction Date, Termination Date), fallback to Termination Date if no extraction
  merged[, End_Date := fcase(
    Event_Observed == 1, End_Event_Date,
    default = Termination_Date
  )]
  
  # Filter out rows where end date is missing or before start date
  merged <- merged[!is.na(Start_Date) & !is.na(End_Date) & End_Date >= Start_Date]
  
  merged[, Survival_Time_Days := as.numeric(End_Date - Start_Date)]
  
  # Select final columns matching the output format
  result <- merged[, .(
    Tooth_ID,
    Start_Date,
    End_Date,
    Survival_Time_Days,
    Event_Observed,
    Index = .I,
    provider_specialty,
    surface_area_code
  )]
  
  return(result)
}
